import React, { useMemo, useId, useRef } from 'react';
import { Star, TrendingUp, Minus, TrendingDown, XCircle } from 'lucide-react';
import { PieChart, Pie, Cell, ResponsiveContainer, Sector } from 'recharts';
import { toPng } from 'html-to-image';

/** --- UTILS --- */
const exportElement = async (ref, fileName) => {
  if (!ref.current) return;
  try {
    const dataUrl = await toPng(ref.current, { quality: 1, pixelRatio: 3, backgroundColor: '#000000', cacheBust: true });
    const link = document.createElement('a');
    link.download = `${fileName}.png`;
    link.href = dataUrl;
    link.click();
  } catch (err) { console.error('Export failed', err); }
};

const getWeekNumber = (date) => {
  const d = new Date(Date.UTC(date.getFullYear(), date.getMonth(), date.getDate()));
  const dayNum = d.getUTCDay() || 7;
  d.setUTCDate(d.getUTCDate() + 4 - dayNum);
  const yearStart = new Date(Date.UTC(d.getUTCFullYear(), 0, 1));
  return Math.ceil((((d.getTime() - yearStart.getTime()) / 86400000) + 1) / 7);
};

const getWeekDates = (date) => {
  const d = new Date(date);
  const diff = d.getDay() === 0 ? -6 : 1 - d.getDay();
  const startDate = new Date(d); startDate.setDate(d.getDate() + diff);
  const endDate = new Date(startDate); endDate.setDate(startDate.getDate() + 6);
  return { startDate, endDate };
};

const formatWeekDateRange = (s, e) => {
  const opt = { month: 'short', day: 'numeric' };
  return s.getMonth() === e.getMonth() 
    ? `${s.toLocaleDateString('en-US', opt)} – ${e.getDate()}, ${e.getFullYear()}`
    : `${s.toLocaleDateString('en-US', opt)} – ${e.toLocaleDateString('en-US', opt)}, ${e.getFullYear()}`;
};

const hexToRgb = (hex) => ({
  r: parseInt(hex.slice(1, 3), 16), g: parseInt(hex.slice(3, 5), 16), b: parseInt(hex.slice(5, 7), 16)
});

const rgbToHsv = ({ r, g, b }) => {
  const rn = r/255, gn = g/255, bn = b/255, max = Math.max(rn, gn, bn), min = Math.min(rn, gn, bn), d = max - min;
  let h = 0, s = max === 0 ? 0 : (d / max) * 100, v = max * 100;
  if (d) {
    if (max === rn) h = ((gn - bn) / d) % 6;
    else if (max === gn) h = (bn - rn) / d + 2;
    else h = (rn - gn) / d + 4;
    h = (h * 60 + 360) % 360;
  }
  return { h, s, v };
};

const hsvToHex = (h, s, v) => {
  const s_n = s/100, v_n = v/100, c = v_n * s_n, x = c * (1 - Math.abs(((h/60) % 2) - 1)), m = v_n - c;
  let r=0, g=0, b=0;
  if (h < 60) [r,g,b]=[c,x,0]; else if (h < 120) [r,g,b]=[x,c,0]; else if (h < 180) [r,g,b]=[0,c,x];
  else if (h < 240) [r,g,b]=[0,x,c]; else if (h < 300) [r,g,b]=[x,0,c]; else [r,g,b]=[c,0,x];
  const toHex = (n) => Math.round((n + m) * 255).toString(16).padStart(2, '0');
  return `#${toHex(r)}${toHex(g)}${toHex(b)}`;
};

const getSaturationBoundedColor = (headerHex, rowHex, count, total) => {
  const hsvH = rgbToHsv(hexToRgb(headerHex));
  const hsvR = rgbToHsv(hexToRgb(rowHex));
  const ratio = total > 0 ? count / total : 0;
  const targetSat = hsvR.s + (hsvH.s - hsvR.s) * ratio;
  return hsvToHex(hsvH.h, targetSat, hsvH.v);
};

const getRankLabel = (s, na) => na ? "N/A" : s>=4.25 ? "TOP" : s>=3.75 ? "POP" : s>=2.5 ? "MOP" : s>=1 ? "FLOP" : "STOP";

/** --- COMPONENTS --- */
const GradientStar = ({ percentage, score, sizeClass="size-7", isNA=false, customGradientStops=null, starColor, onStarClick }) => {
  const id = useId().replace(/:/g,"");
  const fill = Math.max(0, Math.min(100, percentage));
  const activeColor = starColor || '#4B5563';
  const { starPath, facets } = useMemo(() => {
    const pts = [];
    for(let i=0; i<10; i++) {
      const a = (i*Math.PI)/5 - Math.PI/2, r = i%2===0 ? 90 : 40;
      pts.push({x: 100+r*Math.cos(a), y: 100+r*Math.sin(a)});
    }
    return { starPath: `M ${pts[0].x},${pts[0].y} ` + pts.map(p=>`L ${p.x},${p.y}`).join(" ") + " Z", facets: pts.map((p,i)=>`M 100,100 L ${p.x},${p.y} L ${pts[(i+1)%10].x},${pts[(i+1)%10].y} Z`) };
  }, []);

  return (
    <span className={`${sizeClass} inline-flex items-center justify-center relative cursor-pointer`} onClick={onStarClick}>
      <svg width="100%" height="100%" viewBox="0 0 200 200" style={{filter: 'drop-shadow(1px 1px 3px rgba(0,0,0,0.5))'}}>
        <defs>
          <clipPath id={`clip-${id}`}><path d={starPath}/></clipPath>
          <linearGradient id={`grad-${id}`} x1="0%" y1="0%" x2="100%" y2="0%">
            {customGradientStops ? customGradientStops.map((s,i)=><stop key={i} offset={`${s.offset}%`} stopColor={s.color}/>) :
            <><stop offset="0%" stopColor={activeColor}/><stop offset={`${fill}%`} stopColor={activeColor}/><stop offset={`${fill}%`} stopColor="#374151"/><stop offset="100%" stopColor="#374151"/></>}
          </linearGradient>
        </defs>
        <path d={starPath} fill={`url(#grad-${id})`}/>
        <g clipPath={`url(#clip-${id})`}>{facets.map((d,i)=><path key={i} d={d} fill={i%2===0?"white":"black"} fillOpacity={i%2===0?0.07:0.12}/>)}</g>
      </svg>
    </span>
  );
};

const ReleaseItem = ({ title, artist, score, coverLink }) => (
  <div className="relative group w-full aspect-square rounded-xl overflow-hidden bg-white/10 border border-white/20 shadow-lg transition-transform hover:scale-105 active:scale-95 text-left">
    {coverLink && <img src={coverLink} crossOrigin="anonymous" alt="" className="absolute inset-0 w-full h-full object-cover" />}
    <div className="absolute top-0 right-0 bg-black px-2 py-1 rounded-tr-xl rounded-bl-md shadow-xl z-20 border-l border-b border-white/10 text-center">
      <span className="text-[10px] font-extrabold tracking-tighter" style={{color: '#fff'}}>{score.toFixed(2)}</span>
    </div>
    <div className="absolute inset-0 bg-gradient-to-t from-black/95 via-black/40 to-transparent opacity-90"/>
    <div className="absolute bottom-0 left-0 w-full p-2 flex flex-col uppercase font-sans">
      <h3 className="text-white text-[10px] font-black leading-tight truncate">{title}</h3>
      <p className="text-white/70 text-[8px] font-medium leading-tight truncate">{artist}</p>
    </div>
  </div>
);

const TierRow = ({ label, icon: Icon, bgColor, rowBgColor, releases, computedColor, tierRatio }) => {
  const rowRef = useRef(null);
  const avg = releases.length ? releases.reduce((a,r)=>a+r.score,0)/releases.length : 0;
  return (
    <section className="w-full mb-8 font-sans">
      <div ref={rowRef} className="bg-black p-0.5 rounded-3xl">
        <div className="flex flex-col sm:flex-row rounded-2xl overflow-hidden border border-white/10 relative" style={{background: rowBgColor}}>
          <div className="flex flex-col items-center justify-center py-6 sm:w-32 flex-shrink-0 gap-1 text-white cursor-pointer shadow-xl" onClick={()=>exportElement(rowRef, `Tier-${label}`)} style={{background: bgColor}}>
            {Icon && <Icon className="size-7 opacity-90"/>}
            <h2 className="text-2xl font-black uppercase tracking-tighter">{label}</h2>
          </div>
          <div className="flex flex-wrap justify-center p-4 gap-3 flex-grow min-h-36">
            {releases.length ? releases.map((r,i)=><div key={i} className="w-[calc(50%-0.75rem)] sm:w-32 md:w-36"><ReleaseItem {...r}/></div>) : 
            <div className="flex items-center justify-center w-full text-gray-500 font-black uppercase text-xs">NO RELEASES</div>}
          </div>
        </div>
      </div>
      <div className="flex flex-row items-center justify-center p-4 bg-black rounded-2xl border border-white/10 mt-2 gap-8">
        <div className="flex flex-col items-start uppercase font-black tracking-widest text-sm leading-tight">
          <span style={{color: computedColor}}>{label}</span><span className="text-gray-400">RATIO</span>
        </div>
        <div className="flex flex-col items-center gap-1">
          <GradientStar percentage={tierRatio*100} score={avg} sizeClass="size-12" starColor={computedColor} onStarClick={()=>exportElement(rowRef, `Footer-${label}`)}/>
          <span className="text-lg font-black tracking-tighter" style={{color: computedColor}}>{(tierRatio*100).toFixed(2)}%</span>
        </div>
      </div>
    </section>
  );
};

const MusicRankingFooter = ({ avgScore, isNAState, pieData, dominantLabel, dominantColor, tierBase, tierStats }) => {
  const footerRef = useRef(null);
  const customStops = useMemo(() => {
    if (isNAState) return null;
    let curr = 0; const stops = [];
    tierBase.forEach(b => {
      const s = tierStats.find(x => x.label === b.label);
      if (s && s.ratio > 0) {
        stops.push({ offset: (curr * 100).toFixed(2), color: s.computedColor });
        curr += s.ratio;
      }
    });
    if (stops.length) {
      if (parseFloat(stops[0].offset) > 0) stops.unshift({ offset: "0", color: stops[0].color });
      stops.push({ offset: "100", color: stops[stops.length-1].color });
    }
    return stops;
  }, [isNAState, tierStats, tierBase]);

  const gradStyle = customStops ? `linear-gradient(to right, ${customStops.map(s=>`${s.color} ${s.offset}%`).join(', ')})` : 'none';

  return (
    <footer ref={footerRef} className="mt-12 w-full flex flex-col items-center bg-black font-sans pb-16">
      <div className="w-full bg-white/5 border-t border-white/10 px-10 pt-8 flex flex-col items-center min-h-[500px]" style={{clipPath: 'polygon(5% 0%, 95% 0%, 100% 100%, 0% 100%)'}}>
        <div className="w-80 h-80 relative">
          <ResponsiveContainer width="100%" height="100%">
            <PieChart><Pie data={isNAState?[{v:1}]:pieData.filter(d=>d.value>0)} innerRadius={0} outerRadius="80%" dataKey="value" stroke="none">
              {pieData.map((e,i)=><Cell key={i} fill={e.color}/>)}
            </Pie></PieChart>
          </ResponsiveContainer>
        </div>
        <div className="mt-8 flex flex-col items-center gap-4">
          <span className="text-xl font-black uppercase tracking-widest text-gray-500">WEEK RANK</span>
          <div className="flex flex-row items-center gap-8">
            <div className="flex flex-col items-center">
              <GradientStar percentage={100} score={avgScore} sizeClass="size-24" customGradientStops={customStops} starColor={dominantColor} onStarClick={()=>exportElement(footerRef, 'Full-Rank')}/>
              {!isNAState && <span className="text-3xl font-black" style={{backgroundImage: gradStyle, WebkitBackgroundClip: 'text', color: 'transparent'}}>{avgScore.toFixed(2)}</span>}
            </div>
            <span className="text-8xl font-black uppercase" style={{backgroundImage: gradStyle, WebkitBackgroundClip: 'text', color: 'transparent'}}>{getRankLabel(avgScore, isNAState)}</span>
          </div>
        </div>
      </div>
    </footer>
  );
};

const FreshPackOFlow = () => {
  const week = useMemo(() => { const { startDate, endDate } = getWeekDates(new Date()); return { n: getWeekNumber(startDate), d: formatWeekDateRange(startDate, endDate) }; }, []);
  const releases = useMemo(() => [
    {title: 'Бабки', artist: 'OG Buda', score: 5, coverLink: ''},
    {title: 'Взрослеем', artist: 'Illumate', score: 2.36, coverLink: ''},
    {title: 'Лето-жара', artist: 'Кравц', score: 3.5, coverLink: ''},
    {title: 'SURF & TURF', artist: 'KOOL SAVAS', score: 4.8, coverLink: ''},
    {title: 'НЕДОСТУПНА', artist: 'ВАНЯ ДМИТРИЕНКО', score: 4.37, coverLink: ''},
    {title: 'ШИПЫ И РОЗЫ', artist: 'YANIX', score: 2.9, coverLink: ''},
    {title: 'ЛУЖНИКИ 2026', artist: 'БАСТА, GUF', score: 4.67, coverLink: ''}
  ].sort((a,b)=>b.score-a.score), []);

  const tierBase = useMemo(() => [
    {label: "TOP", icon: Star, bgColor: "#059669", rowBgColor: "#05966966"},
    {label: "POP", icon: TrendingUp, bgColor: "#7d9a3a", rowBgColor: "#7d9a3a66"},
    {label: "MOP", icon: Minus, bgColor: "#f59e0b", rowBgColor: "#f59e0b66"},
    {label: "FLOP", icon: TrendingDown, bgColor: "#ea580c", rowBgColor: "#ea580c66"},
    {label: "STOP", icon: XCircle, bgColor: "#ef4444", rowBgColor: "#ef444466"}
  ], []);

  const metrics = useMemo(() => {
    const grouped = {TOP:[], POP:[], MOP:[], FLOP:[], STOP:[]};
    releases.forEach(r => grouped[getRankLabel(r.score, false)].push(r));
    const stats = tierBase.map(t => {
      const rels = grouped[t.label];
      const color = getSaturationBoundedColor(t.bgColor, t.rowBgColor, rels.length, releases.length);
      return { label: t.label, ratio: releases.length ? rels.length/releases.length : 0, rels, color };
    });
    const dominant = stats.reduce((p,c) => c.rels.length > p.rels.length ? c : p, stats[0]);
    return { grouped, stats, dominant, avg: releases.reduce((a,b)=>a+b.score,0)/releases.length };
  }, [releases, tierBase]);

  return (
    <div className="min-h-screen bg-black text-white p-4 sm:p-8 font-sans">
      <div className="max-w-5xl mx-auto">
        <header className="flex flex-col items-center mb-10 text-center">
          <h1 className="text-5xl font-black tracking-tighter uppercase bg-gradient-to-r from-green-500 to-red-600 bg-clip-text text-transparent">FRESH PACK O'FLOW</h1>
          <p className="text-xl font-bold text-gray-400">{week.d} | WEEK {week.n}</p>
        </header>
        <main className="space-y-4">
          {tierBase.map((t, i) => (
            <TierRow key={t.label} {...t} releases={metrics.grouped[t.label]} tierRatio={metrics.stats[i].ratio} computedColor={metrics.stats[i].color} />
          ))}
        </main>
        <MusicRankingFooter avgScore={metrics.avg} isNAState={!releases.length} pieData={metrics.stats.map(s=>({name: s.label, value: s.rels.length, color: s.color}))} dominantLabel={metrics.dominant.label} dominantColor={metrics.dominant.color} tierBase={tierBase} tierStats={metrics.stats.map(s=>({label: s.label, ratio: s.ratio, computedColor: s.color}))} />
      </div>
    </div>
  );
};

export default FreshPackOFlow;
